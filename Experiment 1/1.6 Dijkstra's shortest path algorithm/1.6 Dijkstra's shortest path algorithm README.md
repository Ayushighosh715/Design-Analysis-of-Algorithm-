## Problem Statement ##
Given a graph s and source vertex s, Dijkstra's shortest path algorithm is used to find the shortest paths from source s to all vertices in the given graph.
-----
## algorithm ##
1.Read the number of vertices, edges, and graph data.
2.Read the source vertex.
3.Initialize distance[] to INFINITY, visited[] to 0, and parent[] to -1.
4.Set distance[source] = 0.
5.Select the unvisited vertex having the smallest distance.
6.Mark the selected vertex as visited.
7.For each adjacent vertex, update its distance if a shorter path is found:
 - distance[v] = distance[u] + weight(u,v)
 - Set parent[v] = u.
8.Repeat steps 5–7 until all reachable vertices are processed.
9.For every vertex, display its shortest distance and trace the path using parent[].
10.If a vertex is unreachable, display INF / NO PATH.
---

##Code##
#include <limits.h> 
#include <stdio.h>
#define MAX 20
int V, E;
int graph[MAX][MAX];
#define INFINITY 99999

void dijkstra(int G[MAX][MAX], int n, int startnode) {
	
	// write your code here
	
	// write your code here
int distance[MAX], visited[MAX], parent[MAX];
int i, j, min, next;
int path[MAX], count, current;

/* Initialize */
for (i = 1; i <= n; i++) {
    distance[i] = INFINITY;
    visited[i] = 0;
    parent[i] = -1;
}

distance[startnode] = 0;

/* Dijkstra */
for (i = 1; i <= n; i++) {
    min = INFINITY;
    next = -1;

    for (j = 1; j <= n; j++) {
        if (!visited[j] && distance[j] < min) {
            min = distance[j];
            next = j;
        }
    }

    if (next == -1)
        break;

    visited[next] = 1;

    for (j = 1; j <= n; j++) {
        if (j != next && G[next][j] != 0) {
            if (distance[next] + G[next][j] < distance[j]) {
                distance[j] = distance[next] + G[next][j];
                parent[j] = next;
            }
        }
    }
}

/* Output */
printf("Node\tDistance\tPath\n");

for (i = 1; i <= n; i++) {

    /* Do not print source vertex */
    if (i == startnode)
        continue;

    printf("%4d\t", i);

    /* Unreachable vertex */
    if (distance[i] == INFINITY) {
        printf("     INF\tNO PATH\n");
    }
    else {
        printf("%8d\t", distance[i]);

        /* Build path: destination -> source */
        current = i;
        count = 0;

        while (current != -1) {
            path[count++] = current;
            current = parent[current];
        }

        for (j = 0; j < count; j++) {
            printf("%d", path[j]);

            if (j < count - 1)
                printf("<-");
        }

        printf("\n");
    }
}




}
